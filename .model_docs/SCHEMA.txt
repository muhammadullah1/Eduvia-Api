module.exports = (sequelize, DataTypes) => {
  const Absence = sequelize.define(“absence”, {
    id: {
      allowNull: false,
      autoIncrement: true,
      primaryKey: true,
      type: DataTypes.INTEGER,
    }
    fk_attendance_id: {
      type: DataTypes.INTEGER,
      allowNull: false,
    },
    absent_details: {
      type: DataTypes.STRING,
      allowNull: true,
    },
    archive: {
      type: DataTypes.BOOLEAN,
      allowNull: false,
      defaultValue: false,
    },
    createdAt: {
      allowNull: false,
      type: DataTypes.INTEGER,
    },
    updatedAt: {
      allowNull: false,
      type: DataTypes.INTEGER,
    },
  });
  Absence.beforeCreate((report) => {
    report.dataValues.createdAt = moment().utc().unix();
    report.dataValues.updatedAt = moment().utc().unix();
  });
  Absence.beforeUpdate((report) => {
    report.dataValues.updatedAt = moment().utc().unix();
  });

  Absence.associate = (models) => {
    Absence.belongsTo(models.Attendance, {
      foreignKey: “fk_attendance_id”,
      as: “attendance”,
    });
  };

  return Absence;
};
